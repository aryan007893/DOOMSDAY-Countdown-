import React, { useState, useEffect, useRef } from 'react';

const DoomsdayWidget = () => {
  // Target date: December 18, 2026, 05:00 UTC (Adjusted +18 seconds for exact stream sync)
  const TARGET_DATE = new Date('2026-12-18T05:00:00Z').getTime() + 18000;

  const [timeLeft, setTimeLeft] = useState({
    months: 0,
    days: 0,
    hours: 0,
    minutes: 0,
    seconds: 0,
  });

  const [isLoaded, setIsLoaded] = useState(false);
  const [isMuted, setIsMuted] = useState(true);
  
  // Web Audio API refs for perfect gapless playback
  const audioCtxRef = useRef(null);
  const audioBufferRef = useRef(null);
  const sourceNodeRef = useRef(null);

  const initAudioContext = () => {
    if (!audioCtxRef.current) {
      audioCtxRef.current = new (window.AudioContext || window.webkitAudioContext)();
    }
  };

  const playGaplessAudio = () => {
    if (!audioCtxRef.current || !audioBufferRef.current) return;
    
    // Stop any existing source node before starting a new one
    if (sourceNodeRef.current) {
      try { sourceNodeRef.current.stop(); } catch (e) {}
      sourceNodeRef.current.disconnect();
    }

    const source = audioCtxRef.current.createBufferSource();
    source.buffer = audioBufferRef.current;
    source.loop = true; // Web Audio API looping is completely gapless
    source.connect(audioCtxRef.current.destination);
    source.start(0);
    sourceNodeRef.current = source;
  };

  const handleAudioUpload = (event) => {
    const file = event.target.files[0];
    if (file) {
      const reader = new FileReader();
      reader.onload = async (e) => {
        const arrayBuffer = e.target.result;
        initAudioContext();
        
        try {
          const decodedData = await audioCtxRef.current.decodeAudioData(arrayBuffer);
          audioBufferRef.current = decodedData;
          
          playGaplessAudio();
          
          if (audioCtxRef.current.state === 'suspended') {
            audioCtxRef.current.resume();
          }
          setIsMuted(false);
        } catch (error) {
          console.error("Error decoding audio data for gapless playback:", error);
        }
      };
      reader.readAsArrayBuffer(file);
    }
  };

  const toggleMute = () => {
    initAudioContext();
    
    if (isMuted) {
      if (audioCtxRef.current.state === 'suspended') {
        audioCtxRef.current.resume();
      } else if (!sourceNodeRef.current && audioBufferRef.current) {
        playGaplessAudio();
      }
    } else {
      audioCtxRef.current.suspend(); // Pauses the audio perfectly
    }

    setIsMuted(!isMuted);
  };

  useEffect(() => {
    // Cleanup audio on unmount
    return () => {
      if (sourceNodeRef.current) {
        try { sourceNodeRef.current.stop(); } catch (e) {}
        sourceNodeRef.current.disconnect();
      }
      if (audioCtxRef.current && audioCtxRef.current.state !== 'closed') {
        audioCtxRef.current.close();
      }
    };
  }, []);

  useEffect(() => {
    // Inject Google Font for that cinematic digital look
    const link = document.createElement('link');
    link.href = 'https://fonts.googleapis.com/css2?family=Orbitron:wght@500;700;900&display=swap';
    link.rel = 'stylesheet';
    document.head.appendChild(link);

    const updateCountdown = () => {
      const targetDateObj = new Date(TARGET_DATE);
      const now = new Date();

      if (targetDateObj.getTime() <= now.getTime()) {
        setTimeLeft({ months: 0, days: 0, hours: 0, minutes: 0, seconds: 0 });
        setIsLoaded(true);
        return;
      }

      // Calculate exact months between dates
      let months = (targetDateObj.getFullYear() - now.getFullYear()) * 12 + (targetDateObj.getMonth() - now.getMonth());
      let tempDate = new Date(now.getTime());
      tempDate.setMonth(tempDate.getMonth() + months);

      // If adding the months overshoots the target, drop back by one month
      if (tempDate.getTime() > targetDateObj.getTime()) {
        months--;
        tempDate = new Date(now.getTime());
        tempDate.setMonth(tempDate.getMonth() + months);
      }

      // Calculate remaining time after subtracting the full months
      const difference = targetDateObj.getTime() - tempDate.getTime();

      setTimeLeft({
        months: months,
        days: Math.floor(difference / (1000 * 60 * 60 * 24)),
        hours: Math.floor((difference % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60)),
        minutes: Math.floor((difference % (1000 * 60 * 60)) / (1000 * 60)),
        seconds: Math.floor((difference % (1000 * 60)) / 1000),
      });
      setIsLoaded(true);
    };

    updateCountdown();
    const interval = setInterval(updateCountdown, 1000);

    return () => clearInterval(interval);
  }, []);

  // Format numbers to always be 2 digits and ensure string format for splitting
  const formatNumber = (num) => (num < 10 ? `0${num}` : String(num));

  // Helper function to render independent animated digits
  const renderDigits = (value, label) => {
    return formatNumber(value).split('').map((digit, index) => (
      <span key={`${label}-${index}-${digit}`} className="animate-flip inline-block">
        {digit}
      </span>
    ));
  };

  return (
    <div className="min-h-screen w-full bg-black flex flex-col items-center justify-center p-4 overflow-hidden relative">
      {/* Background Image */}
      <img 
        src="50343_2.jpg"
        alt="Doctor Doom Background"
        style={{
          position: 'absolute',
          top: 0,
          left: 0,
          width: '100%',
          height: '100%',
          objectFit: 'cover',
          zIndex: 0,
          opacity: 1
        }}
      />

      {/* Custom Styles for Glow and Animations */}
      <style>{`
        body {
          margin: 0;
          background-color: #000;
        }
        .font-orbitron {
          font-family: 'Orbitron', sans-serif;
        }
        .doom-glow {
          color: #39FF14;
          text-shadow: 0 0 10px #39FF14, 0 0 20px #39FF14, 0 0 40px rgba(57, 255, 20, 0.4);
          animation: pulseGlow 2s ease-in-out infinite alternate;
          will-change: opacity;
        }
        @keyframes pulseGlow {
          from { opacity: 0.85; }
          to { opacity: 1; }
        }
        .scanline {
          width: 100%;
          height: 100px;
          background: linear-gradient(0deg, rgba(0,0,0,0) 0%, rgba(57,255,20,0.1) 50%, rgba(0,0,0,0) 100%);
          opacity: 0.1;
          position: absolute;
          top: -100px;
          animation: scanline 8s linear infinite;
          pointer-events: none;
          will-change: transform;
        }
        @keyframes scanline {
          0% { transform: translateY(0); }
          100% { transform: translateY(120vh); }
        }
        .vignette {
          background: radial-gradient(circle, rgba(0,0,0,0) 40%, rgba(0,0,0,0.7) 100%);
          pointer-events: none;
        }
        @keyframes flipIn {
          0% { transform: rotateX(90deg); opacity: 0; }
          100% { transform: rotateX(0deg); opacity: 1; }
        }
        .animate-flip {
          animation: flipIn 0.4s ease-out forwards;
          display: inline-block;
          transform-style: preserve-3d;
          backface-visibility: hidden;
          will-change: transform, opacity;
        }
      `}</style>

      {/* Background Effects */}
      <div className="absolute inset-0 vignette z-10"></div>
      <div className="scanline z-10"></div>

      {/* Sound Controls */}
      <div className="absolute top-6 right-6 z-50 flex gap-3">
        {/* Upload Audio Button */}
        <label 
          className="text-[#39FF14] border border-[#39FF14] bg-black/50 p-3 rounded-full hover:bg-[#39FF14]/20 transition-all doom-glow cursor-pointer flex items-center justify-center"
          title="Upload Custom Sound"
        >
          <input type="file" accept="audio/*" className="hidden" onChange={handleAudioUpload} />
          <svg xmlns="http://www.w3.org/2000/svg" width="24" height="24" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"><path d="M21 15v4a2 2 0 0 1-2 2H5a2 2 0 0 1-2-2v-4"></path><polyline points="17 8 12 3 7 8"></polyline><line x1="12" y1="3" x2="12" y2="15"></line></svg>
        </label>

        {/* Sound Toggle Button */}
        <button 
          onClick={toggleMute}
          className="text-[#39FF14] border border-[#39FF14] bg-black/50 p-3 rounded-full hover:bg-[#39FF14]/20 transition-all doom-glow cursor-pointer"
          aria-label="Toggle Sound"
        >
          {isMuted ? (
            <svg xmlns="http://www.w3.org/2000/svg" width="24" height="24" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"><polygon points="11 5 6 9 2 9 2 15 6 15 11 19 11 5"></polygon><line x1="23" y1="9" x2="17" y2="15"></line><line x1="17" y1="9" x2="23" y2="15"></line></svg>
          ) : (
            <svg xmlns="http://www.w3.org/2000/svg" width="24" height="24" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"><polygon points="11 5 6 9 2 9 2 15 6 15 11 19 11 5"></polygon><path d="M15.54 8.46a5 5 0 0 1 0 7.07"></path><path d="M19.07 4.93a10 10 0 0 1 0 14.14"></path></svg>
          )}
        </button>
      </div>

      {/* Main Content */}
      <div className={`relative z-20 flex flex-col items-center justify-center w-full px-2 gap-4 md:gap-8 transition-opacity duration-1000 ${isLoaded ? 'opacity-100' : 'opacity-0'}`}>
        
        {/* Title */}
        <h1 className="text-lg sm:text-xl md:text-3xl lg:text-5xl font-orbitron font-black tracking-widest text-center uppercase doom-glow leading-tight w-full">
          DOOMSDAY IS COMING
        </h1>

        {/* Countdown Container */}
        <div className="flex flex-row flex-nowrap items-center justify-center gap-2 sm:gap-3 md:gap-5 lg:gap-8 w-full max-w-5xl">
          
          {/* Months */}
          <div className="flex flex-col items-center">
            <div className="text-lg md:text-2xl lg:text-4xl font-orbitron font-bold doom-glow tabular-nums tracking-tighter flex">
              {renderDigits(timeLeft.months, 'months')}
            </div>
            <div className="text-[#39FF14] font-orbitron text-[5.5px] md:text-[7.5px] lg:text-[9px] uppercase tracking-[0.2em] md:tracking-[0.4em] mt-2 opacity-80">
              Months
            </div>
          </div>

          {/* Separator */}
          <div className="text-lg md:text-2xl lg:text-4xl font-orbitron font-bold doom-glow opacity-50 mb-4 md:mb-6">
            :
          </div>

          {/* Days */}
          <div className="flex flex-col items-center">
            <div className="text-lg md:text-2xl lg:text-4xl font-orbitron font-bold doom-glow tabular-nums tracking-tighter flex">
              {renderDigits(timeLeft.days, 'days')}
            </div>
            <div className="text-[#39FF14] font-orbitron text-[5.5px] md:text-[7.5px] lg:text-[9px] uppercase tracking-[0.2em] md:tracking-[0.4em] mt-2 opacity-80">
              Days
            </div>
          </div>

          {/* Separator */}
          <div className="text-lg md:text-2xl lg:text-4xl font-orbitron font-bold doom-glow opacity-50 mb-4 md:mb-6">
            :
          </div>

          {/* Hours */}
          <div className="flex flex-col items-center">
            <div className="text-lg md:text-2xl lg:text-4xl font-orbitron font-bold doom-glow tabular-nums tracking-tighter flex">
              {renderDigits(timeLeft.hours, 'hours')}
            </div>
            <div className="text-[#39FF14] font-orbitron text-[5.5px] md:text-[7.5px] lg:text-[9px] uppercase tracking-[0.2em] md:tracking-[0.4em] mt-2 opacity-80">
              Hours
            </div>
          </div>

          {/* Separator */}
          <div className="text-lg md:text-2xl lg:text-4xl font-orbitron font-bold doom-glow opacity-50 mb-4 md:mb-6">
            :
          </div>

          {/* Minutes */}
          <div className="flex flex-col items-center">
            <div className="text-lg md:text-2xl lg:text-4xl font-orbitron font-bold doom-glow tabular-nums tracking-tighter flex">
              {renderDigits(timeLeft.minutes, 'minutes')}
            </div>
            <div className="text-[#39FF14] font-orbitron text-[5.5px] md:text-[7.5px] lg:text-[9px] uppercase tracking-[0.2em] md:tracking-[0.4em] mt-2 opacity-80">
              Minutes
            </div>
          </div>

          {/* Separator */}
          <div className="text-lg md:text-2xl lg:text-4xl font-orbitron font-bold doom-glow opacity-50 mb-4 md:mb-6">
            :
          </div>

          {/* Seconds */}
          <div className="flex flex-col items-center">
            <div className="text-lg md:text-2xl lg:text-4xl font-orbitron font-bold doom-glow tabular-nums tracking-tighter flex">
              {renderDigits(timeLeft.seconds, 'seconds')}
            </div>
            <div className="text-[#39FF14] font-orbitron text-[5.5px] md:text-[7.5px] lg:text-[9px] uppercase tracking-[0.2em] md:tracking-[0.4em] mt-2 opacity-80">
              Seconds
            </div>
          </div>

        </div>

      </div>
    </div>
  );
};

export default DoomsdayWidget;
